// ============================================================ // SISTEMA DE REGISTRO DE ASISTENCIA - v4 (compatible con CORS) // ============================================================ var SHEET_ID = 'TU_SHEET_ID_AQUI'; var HOJA_REGISTROS = 'Registros'; var HORA_ENTRADA_LIMITE = { hora: 9, minuto: 0 }; var HORA_SALIDA_LIMITE = { hora: 18, minuto: 0 }; function doGet(e) { var params = e.parameter; // Si viene con accion=registrar, procesar el registro if (params.accion === 'registrar') { var nombre = params.nombre || ''; var id = params.id || ''; var geoEstado = params.geo || 'sin_validar'; if (!nombre || !id) { var errResp = ContentService.createTextOutput(JSON.stringify({ exito: false, error: 'Parametros invalidos' })); errResp.setMimeType(ContentService.MimeType.JSON); return errResp; } var resultado = registrarAsistencia(nombre, id, geoEstado); var output = ContentService.createTextOutput(JSON.stringify(resultado)); output.setMimeType(ContentService.MimeType.JSON); return output; } // Si viene sin accion, mostrar pagina de error return HtmlService.createHtmlOutput(paginaError('Acceso directo no permitido.')); } function registrarAsistencia(nombre, id, geoEstado) { var ss = SpreadsheetApp.openById(SHEET_ID); var hoja = ss.getSheetByName(HOJA_REGISTROS); var ahora = new Date(); var ultimoRegistro = obtenerUltimoRegistro(hoja, id); var tipo = 'Entrada'; if (ultimoRegistro && ultimoRegistro.tipo === 'Entrada') { tipo = 'Salida'; } var estadoHorario = determinarEstado(ahora, tipo); var estadoFinal = estadoHorario; if (geoEstado === 'sin_validar') { estadoFinal = estadoHorario === 'Normal' ? 'Sin validar ubicacion' : estadoHorario + ' - Sin ubicacion'; } var fecha = Utilities.formatDate(ahora, 'America/Santiago', 'dd/MM/yyyy'); var hora = Utilities.formatDate(ahora, 'America/Santiago', 'HH:mm:ss'); hoja.appendRow([nombre, id, fecha, hora, tipo, estadoFinal]); return { exito: true, nombre: nombre, tipo: tipo, hora: hora, fecha: fecha, estado: estadoFinal }; } function obtenerUltimoRegistro(hoja, id) { var datos = hoja.getDataRange().getValues(); for (var i = datos.length - 1; i >= 1; i--) { if (String(datos[i][1]) === String(id)) { return { nombre: datos[i][0], id: datos[i][1], fecha: datos[i][2], hora: datos[i][3], tipo: datos[i][4], estado: datos[i][5] }; } } return null; } function determinarEstado(ahora, tipo) { var total = ahora.getHours() * 60 + ahora.getMinutes(); if (tipo === 'Entrada') { var lim = HORA_ENTRADA_LIMITE.hora * 60 + HORA_ENTRADA_LIMITE.minuto; return total > lim ? 'Atrasado' : 'Normal'; } else { var lim2 = HORA_SALIDA_LIMITE.hora * 60 + HORA_SALIDA_LIMITE.minuto; return total < lim2 ? 'Salida anticipada' : 'Normal'; } } function paginaError(mensaje) { return '

⚠

' + mensaje + '

'; }
